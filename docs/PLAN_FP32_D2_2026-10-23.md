```python
import threading

import numpy as np
from ase import Atoms

from explore.ledger import Ledger
from explore.parallel import TaskJob, run_many
from test_scan_performance import CoupledBonds, triangle


def evaluator(calls):
    def batch(frames):
        calls.append(len(frames))
        out = []
        for f in frames:
            a = f.copy()
            a.calc = CoupledBonds()
            out.append((a.get_potential_energy(), a.get_forces()))
        return out

    return batch


def energy_job(ledger, positions):
    def run(factory):
        a = triangle()
        a.positions = positions
        a.calc = factory()
        return a.get_potential_energy()

    return TaskJob(run, ledger)


def test_tasks_are_batched_and_each_ledger_pays_its_own(tmp_path):
    calls = []
    jobs = [
        energy_job(Ledger(tmp_path / "l.jsonl", "t", replay=False), triangle().positions + 0.01 * k)
        for k in range(4)
    ]
    outcomes, timing = run_many(jobs, evaluator(calls), max_active=4)
    assert all(o.error is None for o in outcomes)
    assert [j.ledger.spent for j in jobs] == [1, 1, 1, 1]
    assert calls == [4]


def test_hessian_requests_run_alone_on_the_coordinator_and_cost_one(tmp_path):
    seen = []

    def hessian(atoms):
        seen.append(threading.current_thread() is threading.main_thread())
        return np.eye(3 * len(atoms))

    def run(factory):
        a = triangle()
        a.calc = factory()
        a.get_potential_energy()
        return factory().get_hessian(a)

    ledger = Ledger(tmp_path / "l.jsonl", "t", replay=False)
    (out,), _ = run_many([TaskJob(run, ledger)], evaluator([]), 1, hessian=hessian)
    assert out.error is None and out.result.shape == (9, 9)
    assert seen == [True] and ledger.spent == 2


def test_no_hessian_backend_means_no_get_hessian_attribute(tmp_path):
    def run(factory):
        return hasattr(factory(), "get_hessian")

    ledger = Ledger(tmp_path / "l.jsonl", "t", replay=False)
    (out,), _ = run_many([TaskJob(run, ledger)], evaluator([]), 1)
    assert out.result is False


def test_one_failing_task_does_not_stop_the_others(tmp_path):
    def bad(factory):
        raise ValueError("boom")

    good = energy_job(Ledger(tmp_path / "g.jsonl", "t", replay=False), triangle().positions)
    outcomes, _ = run_many(
        [TaskJob(bad, Ledger(tmp_path / "b.jsonl", "t", replay=False)), good],
        evaluator([]),
        2,
    )
    assert outcomes[0].error.startswith("ValueError") and outcomes[1].error is None
```

### Task 4: fp32 下 Hessian 容差的实测与推导（180 GPU）

**Files:** Create `benchmarks/explore_d2/fp32_hessian_noise.py`；结果 `benchmarks/explore_d2/results/<日期>/fp32_hessian_noise/`

**量什么：** 常驻 fp32 POLAR-1，对 9 个结构各测一组：10-22 的 7 个软模结构（读法照抄 `fp32_hessian_softmodes.py`）、b04 `ts0000`、b05 sB 的 E2 鞍点（取自 `results/2026-10-23/b05_d2_sB/journal.jsonl` 里 `c2` 的 `saddle_positions`）。每个结构测：
1. **解析 Hessian 的重复抖动**：同一结构做 5 次随机刚体转动，每次调 PRRS `hessian_spectrum(candidate, factory, config, source="analytic")`，记最低 12 个内部本征值；按模式统计 5 次的标准差 σ_λ 和极差。
2. **力噪声**：同一几何 5 次随机刚体转动，把力转回原坐标系后，统计每个分量的标准差 σ_F（eV/Å）。
3. 用现行 PRRS 默认容差 `confirm_minimum` 一次，记下结论和 reason。

全部 forward 走一个 `Ledger`；fp32；`OMP_NUM_THREADS=4`。

**推导规则（写进脚本输出和 DONE_LOG）：**
- `minimum_check_eigenvalue_tol`（本征值的“显著”门）：取 9 个结构中最低 12 个内部模式 σ_λ 的最大值，乘 3 再取整到一位有效数字；不小于现行 1e-3（不会比现在更紧）。依据：本征值噪声近似正态，3σ 单侧误判率约 0.1%。
- 有限差分步长 `minimum_check_step_A` 与解析核对容差 `analytic_hessian_check`：有限差分的列噪声约为 √2·σ_F / h（中心差分）。取 h 使这个噪声 ≤ 新的本征值容差的 1/3；`analytic_hessian_check`（相对量）不小于“该噪声 ÷ 典型 Hessian-向量积的大小”的 3 倍。若算出的 h 比现行 0.01 Å 小，就保留 0.01。
- `trivial_floor_fraction`、`internal_mode_recheck_step_scale`：本轮不改，只报告它们在 fp32 下触发了几次。
- 推导出的新值只用在 explore 的运行上（脚本里显式传给 `SearchConfig`），不改 PRRS 默认值。
